% Final report preamble, included in the LaTeX header by build.sh.
% Differs from docs/d1/preamble.tex: the report has a 5-page limit, so each
% "#" section (a chapter in the report class) runs on without a page break
% and with a compact heading instead of "Chapter N".

\usepackage{graphicx}
\usepackage{etoolbox}
\usepackage{titlesec}

\titleformat{\chapter}[hang]{\Large\bfseries}{\thechapter}{0.8em}{}
\titlespacing*{\chapter}{0pt}{14pt}{6pt}
\makeatletter
\patchcmd{\chapter}{\if@openright\cleardoublepage\else\clearpage\fi}{\par}{}{}
\makeatother

% Tables in a slightly smaller type, and block paragraphs.
\AtBeginEnvironment{longtable}{\small}
\setlength{\parindent}{0pt}
\setlength{\parskip}{4pt}

% More room above the text on every page.
\geometry{top=2.6cm}
